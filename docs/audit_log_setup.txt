-- Módulo 1: Logs de Auditoria
-- Executar no Supabase SQL Editor

create table public.audit_log (
  id            uuid        primary key default gen_random_uuid(),
  user_id       uuid        references auth.users (id) on delete set null,
  action        text        not null,
  details       jsonb       not null default '{}',
  created_at    timestamptz not null default now()
);

create index audit_log_user_id_idx    on public.audit_log (user_id);
create index audit_log_created_at_idx on public.audit_log (created_at desc);

alter table public.audit_log enable row level security;

-- Usuários só inserem nos próprios logs, não leem (acesso só pelo dashboard)
create policy "audit_log_insert_own" on public.audit_log
  for insert to authenticated
  with check (user_id = auth.uid());

-- Função helper com security definer para garantir o insert
create or replace function public.log_audit(
  p_action  text,
  p_details jsonb default '{}'
)
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.audit_log (user_id, action, details)
  values (auth.uid(), p_action, p_details);
end;
$$;

-- Trigger: log de update em profiles
create or replace function public.trg_log_profile_update()
returns trigger language plpgsql security definer set search_path = '' as $$
begin
  perform public.log_audit(
    'profile_update',
    jsonb_build_object(
      'display_name_changed', (OLD.display_name is distinct from NEW.display_name),
      'weight_changed',       (OLD.weight_kg     is distinct from NEW.weight_kg),
      'goal_changed',         (OLD.goal          is distinct from NEW.goal)
    )
  );
  return NEW;
end;
$$;

create trigger trg_log_profile_update
  after update on public.profiles
  for each row execute function public.trg_log_profile_update();

-- Trigger: log de delete em user_foods
create or replace function public.trg_log_user_food_delete()
returns trigger language plpgsql security definer set search_path = '' as $$
begin
  perform public.log_audit(
    'user_food_delete',
    jsonb_build_object('name', OLD.name)
  );
  return OLD;
end;
$$;

create trigger trg_log_user_food_delete
  before delete on public.user_foods
  for each row execute function public.trg_log_user_food_delete();
